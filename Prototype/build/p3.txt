/* ---------- mulberry32: cùng seed cho cùng dãy số trên mọi máy ---------- */
function Rng(seed){ this.s = seed | 0; }
Rng.prototype.next = function(){
  this.s = (this.s + 0x6D2B79F5) | 0;
  let a = this.s;
  let t = Math.imul(a ^ (a >>> 15), 1 | a);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
Rng.prototype.int = function(n){ return Math.floor(this.next() * n); };
Rng.prototype.range = function(a, b){ return a + Math.floor(this.next() * (b - a + 1)); };

/* Tập số nguyên GIỮ THỨ TỰ CHÈN. Bộ sinh màn duyệt tập ô trống để chọn ô
   "bị bó buộc nhất"; nhiều ô bằng điểm thì kết quả phụ thuộc thứ tự duyệt. */
function OSet(items){
  this.order = []; this.mem = new Set();
  if (items) for (const i of items) this.add(i);
}
OSet.prototype.add = function(v){ if(!this.mem.has(v)){ this.mem.add(v); this.order.push(v); } };
OSet.prototype.del = function(v){
  if (this.mem.delete(v)){ const a = this.order.indexOf(v); if (a >= 0) this.order.splice(a, 1); }
};
OSet.prototype.has = function(v){ return this.mem.has(v); };

/* ================== HÌNH HỌC: kề = 8 hướng ================== */
const DX = [-1,0,1,-1,1,-1,0,1], DY = [-1,-1,-1,0,0,1,1,1];
function makeGeo(cols, rows, active){
  const nb = new Array(cols * rows), cells = [];
  for (let i = 0; i < cols * rows; i++){
    if (!active[i]){ nb[i] = []; continue; }
    cells.push(i);
    const x = i % cols, y = (i / cols) | 0, buf = [];
    for (let d = 0; d < 8; d++){
      const nx = x + DX[d], ny = y + DY[d];
      if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue;
      const j = ny * cols + nx;
      if (active[j]) buf.push(j);
    }
    nb[i] = buf;
  }
  return { cols, rows, active, nb, cells, count: cols * rows };
}

/* ================== PHÂN HOẠCH: dựng NGƯỢC từ lời giải ==================
   Chia cả bàn thành các "đường" kề nhau. Mỗi đường tự nó đã là một chuỗi
   hợp lệ nên xoá được trong đúng 1 lượt => par = số đường là mốc CHẮC CHẮN
   đạt được. Đây là lý do không màn nào bị bất khả thi.                    */
function freeDeg(geo, free, c){ let n = 0; for (const j of geo.nb[c]) if (free.has(j)) n++; return n; }
function sortByDeg(geo, free, arr){        // chèn — ỔN ĐỊNH, thứ tự quyết định bàn
  for (let i = 1; i < arr.length; i++){
    const v = arr[i], k = freeDeg(geo, free, v); let j = i - 1;
    while (j >= 0 && freeDeg(geo, free, arr[j]) > k){ arr[j+1] = arr[j]; j--; }
    arr[j+1] = v;
  }
}
function partition(geo, minLen, maxLen, minChain, maxChain, rng){
  const free = new OSet(geo.cells), paths = [];
  while (free.order.length > 0){
    let start = -1, best = 1e9;
    for (const i of free.order){
      const d = freeDeg(geo, free, i);
      if (d < best){ best = d; start = i; }
      else if (d === best && rng.next() < 0.25) start = i;
    }
    free.del(start);

    const path = [start], target = rng.range(minLen, maxLen);
    while (path.length < target){
      const last = path[path.length - 1], cand = [];
      for (const j of geo.nb[last]) if (free.has(j)) cand.push(j);
      if (!cand.length) break;
      sortByDeg(geo, free, cand);
      const pick = rng.next() < 0.7 ? cand[0] : cand[rng.int(cand.length)];
      free.del(pick); path.push(pick);
    }

    if (path.length < minChain){
      /* Đường quá ngắn để đánh được. Chỉ nối vào ĐẦU hoặc CUỐI một đường có
         sẵn — nối vào giữa thì tập ô mới có thể không xoá được trong 1 lượt. */
      let ok = false;
      for (let p = 0; p < paths.length && !ok; p++){
        const o = paths[p];
        if (o.length + path.length > maxChain) continue;
        if (geo.nb[path[0]].includes(o[o.length-1])){ o.push(...path); ok = true; }
        else if (geo.nb[path[path.length-1]].includes(o[0])){ o.unshift(...path); ok = true; }
      }
      if (!ok) return null;
    } else paths.push(path);
  }
  return paths;
}
/* Mặc định cấm trùng màu với đường kề => mỗi cụm màu là đúng một đường, bàn
   đọc rất rõ. Với xác suất `fuse` thì BỎ lệnh cấm => hai đường kề dính thành
   cục lớn, người chơi phải tự tìm chỗ chẻ. par không đổi.                   */
function colorPaths(geo, paths, colors, fuse, rng){
  const owner = new Array(geo.count).fill(-1);
  paths.forEach((p, i) => p.forEach(c => owner[c] = i));
  const adj = paths.map(() => new OSet());
  paths.forEach((p, i) => p.forEach(c => geo.nb[c].forEach(j => {
    if (owner[j] >= 0 && owner[j] !== i) adj[i].add(owner[j]);
  })));

  const out = new Array(paths.length).fill(-1);
  for (let p = 0; p < paths.length; p++){
    const banned = new Set();
    for (const o of adj[p].order)
      if (out[o] >= 0 && rng.next() >= fuse) banned.add(out[o]);
    let pool = [];
    for (let c = 0; c < colors; c++) if (!banned.has(c)) pool.push(c);
    if (!pool.length) for (let c = 0; c < colors; c++) pool.push(c);
    out[p] = pool[rng.int(pool.length)];
  }
  return out;
}
function pickSome(src, n, rng){
  const pool = src.slice(), out = [];
  while (out.length < n && pool.length) out.push(pool.splice(rng.int(pool.length), 1)[0]);
  return out;
}
